// Navigation bar functionality function updateUserInfo() { const userInfo = sessionStorage.getItem('user_info'); const userIdElement = document.getElementById('userId'); if (userInfo && userIdElement) { const user = JSON.parse(userInfo); userIdElement.textContent = user.id; } else if (userIdElement) { userIdElement.textContent = 'Please Login'; userIdElement.style.cursor = 'pointer'; userIdElement.onclick = function() { // Check if accessed via server if (window.location.protocol === 'http:' || window.location.protocol === 'https:') { window.location.href = '/login'; } else { // If opening file directly, jump to server address window.location.href = 'login.html'; } }; } } // Logout functionality const logoutBtn = document.querySelector('.logout-btn'); if (logoutBtn) { logoutBtn.addEventListener('click', function(e) { e.preventDefault(); if (confirm('Are you sure you want to logout?')) { sessionStorage.removeItem('user_info'); window.location.href = '/login'; } }); } // Initialize navigation bar when page loads document.addEventListener('DOMContentLoaded', updateUserInfo); // Theme switching functionality function initializeThemeSwitch() { const darkModeSwitch = document.getElementById('darkModeSwitch'); if (darkModeSwitch) { // Set initial state const currentTheme = localStorage.getItem('theme') || 'light'; darkModeSwitch.checked = currentTheme === 'dark'; // Bind switch event darkModeSwitch.addEventListener('change', function() { if (window.themeManager) { window.themeManager.setTheme(this.checked ? 'dark' : 'light'); } else { // Backup solution const theme = this.checked ? 'dark' : 'light'; localStorage.setItem('theme', theme); document.documentElement.setAttribute('data-theme', theme); window.dispatchEvent(new CustomEvent('themeChanged', { detail: { theme: theme } })); } }); } } // Initialize theme switching after page load document.addEventListener('DOMContentLoaded', function() { // Immediately apply saved theme const savedTheme = localStorage.getItem('theme') || 'light'; document.documentElement.setAttribute('data-theme', savedTheme); setTimeout(initializeThemeSwitch, 100); // Listen for theme change events window.addEventListener('themeChanged', function(e) { const darkModeSwitch = document.getElementById('darkModeSwitch'); if (darkModeSwitch) { darkModeSwitch.checked = e.detail.theme === 'dark'; } }); });
